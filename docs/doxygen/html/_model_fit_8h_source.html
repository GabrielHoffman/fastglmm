83 // MAX
84 void setDevResids( const vec &dr, const vec &y, const vec &mu){
85
86 // transform from residuals.glm
87 // d.res <- sqrt(pmax((object$family$dev.resids)(y, mu,
88 // wts), 0))
89 // ifelse(y > mu, d.res, -d.res)
90 devianceResiduals = sqrt(pmax(dr, 0));
91 uvec idx = find(y <= mu);
92 devianceResiduals.elem(idx) = -1.0*devianceResiduals.elem(idx);
93 }
94
95 void setFittedValues( const vec &mu_in){

97 }

99 private:
100 void init(){
101 // if model didn't succeed, set values to nan
102 if( ! success ){
103 se.fill(datum::nan);
104 vcov.fill(datum::nan);
105
106 // if vcov is empty, set to be se.n_elem x se.n_elem
107 // with elements nan
108 if( vcov.n_elem == 0){
109 vcov = mat(se.n_elem, se.n_elem);
110 vcov.fill(datum::nan);
111 }
112 }
113 }
