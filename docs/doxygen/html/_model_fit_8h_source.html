117 // MAX
118 void setDevResids(
119 const vec &dr,
120 const vec &y,
121 const vec &mu,
122 const vec w = {}){
123
124 // transform from residuals.glm
125 // d.res <- sqrt(pmax((object$family$dev.resids)(y, mu,
126 // wts), 0))
127 // ifelse(y > mu, d.res, -d.res)
128 devianceResiduals = sqrt(pmax(dr, 0));
129 uvec idx = find(y <= mu);
130 devianceResiduals.elem(idx) = -1.0*devianceResiduals.elem(idx);
131
132 if( ! w.is_empty() ){
133 // if weight is zero, set element to NAN
134 devianceResiduals.elem(find(w == 0.0)).fill(datum::nan);
135 }
136 }
137
138 void setFittedValues( const vec &mu_in, const vec w = {}){
139 mu = mu_in;
140
141 if( ! w.is_empty() ){
142 // if weight is zero, set element to NAN
143 mu.elem(find(w == 0.0)).fill(datum::nan);
144 }
145 }
146
147 private:
148 void init(){
149 // if model didn't succeed, set values to nan
150 if( ! success ){
151 se.fill(datum::nan);
152 vcov.fill(datum::nan);
153
154 // if vcov is empty, set to be se.n_elem x se.n_elem
155 // with elements nan
156 if( vcov.n_elem == 0){
157 vcov = mat(se.n_elem, se.n_elem);
158 vcov.fill(datum::nan);
159 }
160 }
161 }

171 ModelFitGLM( ModelFit &gmf, const string &family, const int &niter ) :
172 ModelFit(gmf), family(family), niter(niter) {
173
174 // extract theta values from "nb:theta"
175 if( regex_search( family, regex("^nb:")) ){
176 string theta_str = regex_replace( family, regex("^nb:"), "");
177 theta = atof(theta_str.c_str());
178 }else{
179 theta = datum::nan;
180 }
181
182 // Account for loss of degrees of freedom when predicted counts are zero in a count model
183 // https://doi.org/10.1515/sagmb-2017-0010
184 // if family is Poisson, quasipoisson or nb
185 // Compute number of entries were predicted values are
186 // effectively zero counts
187 shared_ptr<GLMFamily> fam = getGLMFamily( family );
188 if( fam->isCountModel() ){
189 nZeroPrediction = sum(gmf.mu < 1e-4);
190 }
191 }
192
193 string family = "";
194 double theta = datum::nan;
195 double mu_mean = datum::nan;
196 double y_mean = datum::nan;
197 int niter = 0;
198 double nZeroPrediction = 0;
