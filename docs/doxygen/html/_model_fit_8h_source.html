88 // MAX
89 void setDevResids( const vec &dr, const vec &y, const vec &mu){
90
91 // transform from residuals.glm
92 // d.res <- sqrt(pmax((object$family$dev.resids)(y, mu,
93 // wts), 0))
94 // ifelse(y > mu, d.res, -d.res)
95 devianceResiduals = sqrt(pmax(dr, 0));
96 uvec idx = find(y <= mu);
97 devianceResiduals.elem(idx) = -1.0*devianceResiduals.elem(idx);
98 }
99
100 void setFittedValues( const vec &mu_in){
101 mu = mu_in;
102 }
103
104 private:
105 void init(){
106 // if model didn't succeed, set values to nan
107 if( ! success ){
108 se.fill(datum::nan);
109 vcov.fill(datum::nan);
110
111 // if vcov is empty, set to be se.n_elem x se.n_elem
112 // with elements nan
113 if( vcov.n_elem == 0){
114 vcov = mat(se.n_elem, se.n_elem);
115 vcov.fill(datum::nan);
116 }
117 }
118 }
119};

122class ModelFitGLM : public ModelFit {
123
124 public:
125
126 ModelFitGLM() {}
127
128 ModelFitGLM( ModelFit &gmf, const string &family, const int &niter ) :
129 ModelFit(gmf), family(family), niter(niter) {
130
131 // extract theta values from "nb:theta"
132 if( regex_search( family, regex("^nb:")) ){
133 string theta_str = regex_replace( family, regex("^nb:"), "");
134 theta = atof(theta_str.c_str());
135 }else{
136 theta = datum::nan;
137 }
138
139 // Account for loss of degrees of freedom when predicted counts are zero in a count model
140 // https://doi.org/10.1515/sagmb-2017-0010
141 // if family is Poisson, quasipoisson or nb
142 // Compute number of entries were predicted values are
143 // effectively zero counts
144 shared_ptr<GLMFamily> fam = getGLMFamily( family );
145 if( fam->isCountModel() ){
146 nZeroPrediction = sum(gmf.mu < 1e-4);
147 }
148 }
149
150 string family = "";
151 double theta = datum::nan;
152 double mu_mean = datum::nan;
153 int niter = 0;
154 double nZeroPrediction = 0;
155};
