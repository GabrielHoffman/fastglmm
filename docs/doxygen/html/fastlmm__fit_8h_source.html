170template <typename T1, typename T2, typename T3>
171fastlmm<T1, T2, T3>::fastlmm(
172 const T1 &Y_,
173 const T2 &X_,
174 const spectralDecomp<T3> &dcmp,
175 const vec &weights_,
176 const ModelDetail md,
177 const double &lambda,
178 const bool REML):
179 wsqrt(sqrt(weights_)),
180 Y(Y_ % wsqrt),
181 X(scaleEachCol(X_, wsqrt)),
182 weights(weights_),
183 md(md),
184 lambda(lambda),
185 REML(REML),
186 dcmp(dcmp) {
187
188 this->dcmp.reweight(weights);
189 U = this->dcmp.get_U();
190 s = this->dcmp.get_s();
191
192 n_active = accu(weights != 0.0);
193 Yu = U.t() * Y;
194 Xu = U.t() * X;
195 Gamma_XX = X.t() * X - Xu.t() * Xu;
196 Gamma_XY = X.t() * Y - Xu.t() * Yu;
197 inv_s_delta_Xu = mat( Xu.n_rows, Xu.n_cols);
198}

322template <typename T1, typename T2, typename T3>
323const double fastlmm<T1, T2, T3>::get_rdf() const {
324
325 int n = n_active;
326 int k = s.n_elem;
327
328 // X is already scaled
329 // sum(h1)
330 // h1.sum <- with(object, delta*sum(1/(s+delta))) + (n-k)
331 double h1_sum = delta_hat*(sum(1/(s+delta_hat)) + (n-k) / delta_hat);
332
333 // sum(h2)
334 // A <- with(object, X / delta - U %*% ((s/(delta*s + delta^2)) * crossprod(U, X)))
335 // D <- solve(crossprod(A, X))
336 // h2.sum <- object$delta * sum(A * (A %*% D))
337 vec w = (s/(delta_hat*s + pow(delta_hat,2)));
338 mat A = mat(X / delta_hat - U * scaleEachCol( Xu, w));
339 double h2_sum = delta_hat * trace(solve(A.t() * X, A.t() * A));
340
341 return h1_sum - h2_sum;
342}

345template <typename T1, typename T2, typename T3>
346const vec fastlmm<T1, T2, T3>::hatvalues() const {
347
348 // Usq <- model$U^2
349 T3 Usq = square(U);
350
351 // h1 <- model$delta*with(model, Usq %*% (1/(s+delta))) + (1 - rowSums(Usq))
352 vec h1 = delta_hat * Usq * (1/(s+delta_hat)) + (1 - sum(Usq, 1));
353
354 // A <- with(model, X / delta - U %*% ((s/(delta*s + delta^2)) * crossprod(U, X)))
355 // D <- solve(crossprod(A, X))
356 // h2 <- model$delta * rowSums(A * (A %*% D))
357 vec w = (s/(delta_hat*s + pow(delta_hat,2)));
358 mat A = mat(X / delta_hat - U * scaleEachCol( Xu, w));
359 mat D_A_t = solve(A.t() * X, A.t());
360 vec h2 = delta_hat * sum(A % D_A_t.t(), 1);
361
362 // hatvalues
363 return 1 - h1 + h2;
364}

373template <typename T1, typename T2, typename T3>
374const vec fastlmm<T1, T2, T3>::fitted() const {
375
376 // ** need to scale X because it was transformed at the start
377 // a <- object$U %*% (sqrt(object$s) * ranef.fastlmm(object))
378 // a / sqrt(object$weights) + object$design %*% coef(object)
379 return ((U * (sqrt(s) % blup())) + X * beta) / sqrt(weights);
380}

383template <typename T1, typename T2, typename T3>
384const vec fastlmm<T1, T2, T3>::blup() const {
385
386 // Zw <- c(sqrt(fit$weights)) * fit$Z
387 // A <- crossprod(fit$U, Zw)
388 // A <- with(fit, crossprod(fit$U, c(sqrt(weights)) * U * sqrt(s)))
389 // b <- fit$ru / (fit$s + fit$delta)
390 // crossprod(A, b)
391
392 // T3 A = U.t() * scaleRowsCols(U, sqrt(weights), sqrt(s));
393 // vec b = ru / (s + delta_hat);
394 // return A.t() * b;
395
396 // # since U^T U is identity if the GRM is full rank
397 // v <- with(object, sqrt(s)*ru / (s + delta))
398
399 return (sqrt(s) % ru) / (s + delta_hat);
400}

402template <typename T1, typename T2, typename T3>
403double fastlmm<T1, T2, T3>::ll(const double &delta ) {
404
405 double n = n_active;
406 double rank = Xu.n_rows;
407
408 inv_s_delta = 1 / (s+delta);
409
410 // inv_s_delta_Xu <- inv_s_delta * Xu
411 inv_s_delta_Xu = scaleEachCol(Xu, inv_s_delta);
412
413 // QXX = crossprod(Xu, inv_s_delta_Xu) + Gamma_XX / delta
414 QXX = Xu.t() * inv_s_delta_Xu + Gamma_XX / delta;
415
416 // Ridge penalty
417 // QXX.diag() += lambda;
418 // but not on intercept
419 for( uword i=1; i<QXX.n_rows; ++i){
420 QXX(i,i) += lambda;
421 }
422
423 // QXY = crossprod(Xu, inv_s_delta_Yu) + Gamma_XY / delta
424 QXY = Xu.t() * (inv_s_delta % Yu) + Gamma_XY / delta;
425
426 // beta <<- solve( QXX, QXY)
427 // beta = solve(QXX, QXY, solve_opts::likely_sympd);
428 int status = solve(beta, QXX, QXY,
429 solve_opts::likely_sympd + arma::solve_opts::no_approx);
430
431 // if model failed
432 if( ! status ){
433 // set beta to NAN
434 beta.set_size(QXX.n_rows);
435 beta.fill(datum::nan);
436 }
437
438 // # Eval sig_g
439 // ru <- Yu - Xu %*% beta
440 ru = Yu - Xu * beta;
441
442 // r <- Y - X %*% beta
443 r = Y - X * beta;
444
445 // Qrr <- crossprod(ru, inv_s_delta_ru) + (crossprod(r)[1] - crossprod(ru)[1])/ delta
446 // sig_g <<- Qrr[1] / n
447 double QRR = dot(ru, (inv_s_delta % ru)) + (dot(r,r) - dot(ru,ru)) / delta;
448 sigSq_g = QRR / n;
449
450 // use 2.0 to ensure double precision
451 double logLik = -n/2.0 * log(2.0*M_PI*sigSq_g) - 1.0/2.0 * (sum( log(s + delta ) ) + (n-rank) * log(delta)) - n/2.0;
452
453 // this is fixed, so don't eval every time,
454 // just after estimation
455 // + sum(log(weights))/2.0;
456
457 return logLik;
458}

463static inline double ll_alone_mat( double delta_log, void *arg){
464
465 auto *fit = (fastlmm<mat,mat,mat> *) arg;

487template <typename T1, typename T2, typename T3>
488void fastlmm<T1, T2, T3>::estimate_delta( const double &left, const double &right, const double &tol ){
489
490 double leftIn = left;
491 double rightIn = right;
492 iter = 0;
493
494 // initialize function
495 funcStruct F;
496 F.params = this;
497
498 // Since F.function can't take templated function
499 if( isSpMatrix( U ) ){
500 F.function = & ll_alone_spmat;
501 }else{
502 F.function = & ll_alone_mat;
503 }
504
505 // get maximize log-likelihood
506 // need to mutliply but -1 since it actually minimizes
507 // evaluated at minimum value
508 double res;
509 this->logLik = -1*local_min(leftIn, rightIn, tol, &F, res, iter);
510
511 // augment with value this is constant for varying delta's
512 // weights with zero value, give Inf log values
513 // so use omit_nonfinite
514 this->logLik += sum(omit_nonfinite(log(weights)))/2.0;
515
516 this->delta_hat = exp(res);
517}

552ModelFitLMM fastlmm<T1, T2, T3>::get_result(

561 get_delta(),

568 // res.varFitted = var(fitted());
569

572 // set additional values based on ModelDetail md
573 mat V = this->get_vcov();
574
575 switch( md ){
576 case MAX:
577 // res.hatvalues = hatvalues();
578 case MOST:
579 res.hatvalues = this->hatvalues();
580 case HIGH:
581 res.residuals = this->residuals();
582 case MEDIUM:
583 res.vcov = V;
584 case LOW:
585 res.se = sqrt(diagvec(V));
586 res.rdf = this->get_rdf();
587 case LEAST:
588 break;

591 // if returnUS
592 // return U and s
593 if( returnUS ){
594 switch( dcmp.get_type() ){
595 case GENERAL:

599 // V is identity
600 res.setUS(U, s, eye<sp_mat>(U.n_cols, U.n_cols));
