159template <typename T1, typename T2, typename T3>
160fastlmm<T1, T2, T3>::fastlmm(const T1 &Y_,
161 const T2 &X_,
162 const spectralDecomp<T3> &dcmp,
163 const vec &weights_,
164 const ModelDetail md,
165 const bool REML):
166 wsqrt(sqrt(weights_)),
167 Y(Y_ % wsqrt),
168 X(scaleEachCol(X_, wsqrt)),
169 weights(weights_),
170 md(md),
171 REML(REML),
172 dcmp(dcmp) {
173
174 this->dcmp.reweight(weights);
175 U = this->dcmp.get_U();
176 s = this->dcmp.get_s();
177
178 n_active = accu(weights != 0.0);
179 Yu = U.t() * Y;
180 Xu = U.t() * X;
181 Gamma_XX = X.t() * X - Xu.t() * Xu;
182 Gamma_XY = X.t() * Y - Xu.t() * Yu;
183 inv_s_delta_Xu = mat( Xu.n_rows, Xu.n_cols);
184}

274template <typename T1, typename T2, typename T3>
275const double fastlmm<T1, T2, T3>::get_rdf(){
276
277 int n = n_active;
278 int k = s.n_elem;
279
280 // X is already scaled
281 // sum(h1)
282 // h1.sum <- with(object, delta*sum(1/(s+delta))) + (n-k)
283 double h1_sum = delta_hat*(sum(1/(s+delta_hat)) + (n-k) / delta_hat);
284
285 // sum(h2)
286 // A <- with(object, X / delta - U %*% ((s/(delta*s + delta^2)) * crossprod(U, X)))
287 // D <- solve(crossprod(A, X))
288 // h2.sum <- object$delta * sum(A * (A %*% D))
289 vec w = (s/(delta_hat*s + pow(delta_hat,2)));
290 mat A = mat(X / delta_hat - U * scaleEachCol( Xu, w));
291 mat D_A_t = solve(mat(A.t() * X), A.t());
292 double h2_sum = delta_hat * arma::accu(A % D_A_t.t());
293
294 return h1_sum - h2_sum;
295}

298template <typename T1, typename T2, typename T3>
299const vec fastlmm<T1, T2, T3>::hatvalues(){
300
301 // int n = n_active;
302 // int k = s.n_elem;
303
304 // Usq <- model$U^2
305 T3 Usq = square(U);
306
307 // h1 <- model$delta*with(model, Usq %*% (1/(s+delta))) + (1 - rowSums(Usq))
308 vec h1 = delta_hat * Usq * (1/(s+delta_hat)) + (1 - sum(Usq, 1));
309
310 // A <- with(model, X / delta - U %*% ((s/(delta*s + delta^2)) * crossprod(U, X)))
311 // D <- solve(crossprod(A, X))
312 // h2 <- model$delta * rowSums(A * (A %*% D))
313 vec w = (s/(delta_hat*s + pow(delta_hat,2)));
314 mat A = mat(X / delta_hat - U * scaleEachCol( Xu, w));
315 mat D_A_t = solve(mat(A.t() * X), A.t());
316 vec h2 = delta_hat * sum(A % D_A_t.t(), 1);
317
318 // hatvalues
319 return 1 - h1 + h2;
320}

329template <typename T1, typename T2, typename T3>
330const vec fastlmm<T1, T2, T3>::fitted(){
331
332 // a <- object$U %*% (sqrt(object$s) * ranef.fastlmm(object))
333 // a / sqrt(object$weights) + object$design %*% coef(object)
334 // need to scale X because it was transformed at the start
335 return ((U * (sqrt(s) % blup())) + X * beta) / sqrt(weights);
336}

339template <typename T1, typename T2, typename T3>
340const vec fastlmm<T1, T2, T3>::blup(){
341
342 // Zw <- c(sqrt(fit$weights)) * fit$Z
343 // A <- crossprod(fit$U, Zw)
344 // A <- with(fit, crossprod(fit$U, c(sqrt(weights)) * U * sqrt(s)))
345 // b <- fit$ru / (fit$s + fit$delta)
346 // crossprod(A, b)
347
348 // T3 A = U.t() * scaleRowsCols(U, sqrt(weights), sqrt(s));
349 // vec b = ru / (s + delta_hat);
350 // return A.t() * b;
351
352 // # since U^T U is identity if the GRM is full rank
353 // v <- with(object, sqrt(s)*ru / (s + delta))
354
355 return (sqrt(s) % ru) / (s + delta_hat);
356}

358template <typename T1, typename T2, typename T3>
359double fastlmm<T1, T2, T3>::ll(const double &delta ) {
360
361 double n = n_active;
362 double rank = Xu.n_rows;
363
364 inv_s_delta = 1 / (s+delta);
365
366 // inv_s_delta_Xu <- inv_s_delta * Xu
367 inv_s_delta_Xu = scaleEachCol(Xu, inv_s_delta);
368
369 // QXX = crossprod(Xu, inv_s_delta_Xu) + Gamma_XX / delta
370 QXX = Xu.t() * inv_s_delta_Xu + Gamma_XX / delta;
371
372 // QXY = crossprod(Xu, inv_s_delta_Yu) + Gamma_XY / delta
373 QXY = Xu.t() * (inv_s_delta % Yu) + Gamma_XY / delta;
374
375 // beta <<- solve( QXX, QXY)
376 beta = solve(QXX, QXY, solve_opts::likely_sympd);
377
378 // # Eval sig_g
379 // ru <- Yu - Xu %*% beta
380 ru = Yu - Xu * beta;
381
382 // r <- Y - X %*% beta
383 r = Y - X * beta;
384
385 // Qrr <- crossprod(ru, inv_s_delta_ru) + (crossprod(r)[1] - crossprod(ru)[1])/ delta
386 // sig_g <<- Qrr[1] / n
387 double QRR = dot(ru, (inv_s_delta % ru)) + (dot(r,r) - dot(ru,ru)) / delta;
388 sigSq_g = QRR / n;
389
390 // use 2.0 to ensure double precision
391 double logLik = -n/2.0 * log(2.0*M_PI*sigSq_g) - 1.0/2.0 * (sum( log(s + delta ) ) + (n-rank) * log(delta)) - n/2.0;
392
393 // this is fixed, so don't eval every time,
394 // just after estimation
395 // + sum(log(weights))/2.0;
396
397 return logLik;
398}

427template <typename T1, typename T2, typename T3>
428void fastlmm<T1, T2, T3>::estimate_delta( const double &left, const double &right, const double &tol ){
429
430 double leftIn = left;
431 double rightIn = right;
432 iter = 0;
433
434 // initialize function
435 funcStruct F;
436 F.params = this;
437
438 // Since F.function can't take templated function
439 if( isSpMatrix( U ) ){
440 F.function = & ll_alone_spmat;
441 }else{
442 F.function = & ll_alone_mat;
443 }
444
445 // get maximize log-likelihood
446 // need to mutliply but -1 since it actually minimizes
447 // evaluated at minimum value
448 double res;
449 logLik = -1*local_min(leftIn, rightIn, tol, &F, res, iter);
450
451 // augment with value this is constant for varying delta's
452 // weights with zero value, give Inf log values
453 // so use omit_nonfinite
454 logLik += sum(omit_nonfinite(log(weights)))/2.0;
455
456 delta_hat = exp(res);
457}

469void fastlmm<T1, T2, T3>::update_response(const T1 &Y_,
470 const vec &weights_,
471 const mat &Yu_){

473 // indicator_decomp
474 // modiy this->U and this->s internally
475 // compute sqrt(weights) for

492ModelFitLMM fastlmm<T1, T2, T3>::get_result(

534 switch( dcmp.get_type() ){

540 res.setUS(U, s, eye<sp_mat>(U.n_cols, U.n_cols));
541 break;
542 };
543 }
