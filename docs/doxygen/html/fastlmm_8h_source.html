277template <typename T1, typename T2, typename T3>
278double fastlmm<T1, T2, T3>::ll(const double &delta ) {
279
280 double n = X.n_rows;
281 double rank = Xu.n_rows;
282
283 inv_s_delta = 1 / (s+delta);
284
285 // inv_s_delta_Xu <- inv_s_delta * Xu
286 // R recycles over each column,
287 // here do manually
288 // inv_s_delta_Xu( Xu.n_rows, Xu.n_cols);
289 for(int i=0; i<Xu.n_cols; i++){
290 inv_s_delta_Xu.col(i) = inv_s_delta % Xu.col(i);
291 };
292
293 // QXX = crossprod(Xu, inv_s_delta_Xu) + cp_X_low / delta
294 QXX = Xu.t() * inv_s_delta_Xu + cp_X_low / delta;
295
296 // QXY = crossprod(Xu, inv_s_delta_Yu) + cp_X_low_Y_low / delta
297 QXY = Xu.t() * (inv_s_delta % Yu) + cp_X_low_Y_low / delta;
298
299 // beta <<- solve( QXX, QXY)
300 beta = solve(QXX, QXY, solve_opts::likely_sympd);
301
302 // # Eval sig_g
303 // ru <- Yu - Xu %*% beta
304 ru = Yu - Xu * beta;
305
306 // r <- Y - X %*% beta
307 r = Y - X * beta;
308
309 // Qrr <- crossprod(ru, inv_s_delta_ru) + (crossprod(r)[1] - crossprod(ru)[1])/ delta
310 // sig_g <<- Qrr[1] / n
311 double QRR = dot(ru, (inv_s_delta % ru)) + (dot(r,r) - dot(ru,ru)) / delta;
312 sigSq_g = QRR / n;
313
314 // use 2.0 to ensure double precision
315 double logLik = -n/2.0 * log(2.0*M_PI*sigSq_g) - 1.0/2.0 * (sum( log(s + delta ) ) + (n-rank) * log(delta)) - n/2.0;
316
317 // this is fixed, so don't eval every time,
318 // just after estimation
319 // + sum(log(weights))/2.0;
320
321 return logLik;
322}

351template <typename T1, typename T2, typename T3>
352void fastlmm<T1, T2, T3>::estimate_delta( const double &left, const double &right, const double &tol ){
353
354 double leftIn = left;
355 double rightIn = right;
356 iter = 0;
357
358 // initialize function
359 funcStruct F;
360 F.params = this;
361
362 // Since F.function can't take templated function
363 if( isSpMatrix( U ) ){
364 F.function = & ll_alone_spmat;
365 }else{
366 F.function = & ll_alone_mat;
367 }
368
369 // get maximize log-likelihood
370 // need to mutliply but -1 since it actually minimizes
371 // evaluated at minimum value
372 double res;
373 logLik = -1*local_min(leftIn, rightIn, tol, &F, res, iter);
374
375 // augment with value this is constant for varying delta's
376 logLik += sum(log(weights))/2.0;
377
378 delta_hat = exp(res);
379}
