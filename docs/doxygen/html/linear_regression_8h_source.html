93static ModelFit lm(const arma::mat& X, const arma::colvec& y, const ModelDetail md = LOW, const double &rdf_offset = 0, LMWork *work = nullptr) {
94
95 int n = X.n_rows, k = X.n_cols;
96
97 // allocate work, if not already alloc'd
98 bool alloc_local = false;
99 if( work == nullptr ){
100 alloc_local = true;
101 work = new LMWork();
102 }
103
104 // QR decomp
105 qr_econ(work->Q, work->R, X);
106
107 // back solve
108 vec beta = solve(work->R, trans(work->Q) * y);
109
110 // residuals
111 work->residuals = y - X*beta;
112
113 // std.errors of coefficients
114 double rdf = n - k - rdf_offset;
115 double s2 = dot(work->residuals, work->residuals) / rdf;
116
117 // V = solve(t(R)*R)
118 bool success = inv_sympd(work->V, trans(work->R)*work->R);
119
120 vec stderr;
121 if( success ){
122 stderr = sqrt(s2 * diagvec(work->V));
123 }else{
124 stderr = vec(k, fill::value(datum::nan));
125 beta.fill(datum::nan);
126 }
127
128 // return results with specified level of detail
129 ModelFit fit;
130 switch( md ){
131 case LOW:
132 fit = ModelFit( beta, stderr, s2, rdf);
133 break;
134
135 case MEDIUM:
136 fit = ModelFit( beta, stderr, s2, rdf, work->V * s2);
137 break;
138
139 case HIGH:
140 fit = ModelFit( beta, stderr, s2, rdf, work->V * s2, work->residuals);
141 break;
142
143 case MOST:
144 vec hatvalues = diagvec(work->Q * trans(work->Q));
145 fit = ModelFit( beta, stderr, s2, rdf, work->V * s2, work->residuals, hatvalues);
146 break;
147 }
148
149 // free work if allocated in this function
150 if( alloc_local) delete work;
151
152 return fit;
153}

163template <typename T>
164static tuple<vec, T> preprojection(const vec &y, const mat &X_design, const T &X_features, const vec &weights = {}){
165
166 // naive calculation
167 // vec y_proj = y - X_design * inv(trans(X_design) * X_design) * trans(X_design) * y;
168 // mat X_proj = X_features - X_design * inv(trans(X_design) * X_design) * trans(X_design) * X_features;
169
170 // if weights is empty, set w to ones
171 vec w;
172 if( ! weights.is_empty() ){
173 w = weights;
174 }else{
175 w = vec(y.n_elem).ones();
176 }
177 arma::colvec wsqrt = sqrt(w / mean(w));
178
179 // apply weights in computation to y, X_design, and X_features
180 mat X_design_wsqrt = X_design.each_col() % wsqrt;
181 vec y_wsqrt = y % wsqrt;
182 T X_features_wsqrt = scaleEachCol(X_features, wsqrt);
183
184 // Use QR decomp of X_design, and recycle pre-computed values
185 mat Q, R;
186 qr_econ(Q, R, X_design_wsqrt);
187
188 // back solve
189 vec beta = solve(R, trans(Q) * y_wsqrt);
190 vec y_proj = y_wsqrt - X_design_wsqrt * beta;
191
192 // back solve
193 // use constructor T() to subtract matricies of the same type
194 mat gamma = solve(R, trans(Q) * X_features_wsqrt);
195 T X_proj;
196
197 // cast X_design_wsqrt * gamma to type T if needed
198 if( is_same_v<decltype(X_features_wsqrt), decltype(X_design_wsqrt)> ){
199 X_proj = X_features_wsqrt - X_design_wsqrt * gamma;
200 }else{
201 X_proj = X_features_wsqrt - T(X_design_wsqrt * gamma);
202 }
203
204 return {y_proj, X_proj};
205}

215template <typename T>
216static tuple<vec, T> preprojection(const vec &y, const sp_mat &X_design, const T &X_features, const vec &weights = {}){
217
218 // naive calculation
219 // vec y_proj = y - X_design * inv(trans(X_design) * X_design) * trans(X_design) * y;
220 // mat X_proj = X_features - X_design * inv(trans(X_design) * X_design) * trans(X_design) * X_features;
221
222 // if weights is empty, set w to ones
223 vec w;
224 if( ! weights.is_empty() ){
225 w = weights;
226 }else{
227 w = vec(y.n_elem).ones();
228 }
229 arma::colvec wsqrt = sqrt(w / mean(w));
230
231 // apply weights in computation to y, X_design, and X_features
232 sp_mat X_design_wsqrt = scaleEachCol(X_design, wsqrt);
233 vec y_wsqrt = y % wsqrt;
234 T X_features_wsqrt = scaleEachCol(X_features, wsqrt);
235
236 // recycle sparse crossprod
237 // spsolve uses lapack after converting V to dense matrix
238 // minimal penalty practical V dimensions
239 bool success;
240 vec beta;
241 mat gamma;
242
243 sp_mat V = trans(X_design_wsqrt) * X_design_wsqrt;
244 success = spsolve(beta, V, vec(trans(X_design_wsqrt) * y_wsqrt), "lapack");
245 if( ! success ) beta.fill(datum::nan);
246 vec y_proj = y_wsqrt - X_design_wsqrt * beta;
247
248 success = spsolve(gamma, V, mat(trans(X_design_wsqrt) * X_features_wsqrt), "lapack");
249 if( ! success ) gamma.fill(datum::nan);
250
251 T X_proj;
252 // cast X_design_wsqrt * gamma to type T if needed
253 if( is_same_v<decltype(X_features_wsqrt), decltype(X_design_wsqrt)> ){
254 X_proj = X_features_wsqrt - X_design_wsqrt * gamma;
255 }else{
256 X_proj = X_features_wsqrt - T(X_design_wsqrt * gamma);
257 }
258
259 // Use LU decomp of X_design
260 // requires SuperLU library
261 // -L/opt/homebrew/Cellar/superlu/7.0.0/lib -lsuperlu
262 // but not portable
263 // spsolve_factoriser SF;
264
265 // bool status = SF.factorise(X_design_wsqrt);
266
267 // vec beta;
268 // SF.solve(beta, y_wsqrt);
269 // vec y_proj = y_wsqrt - X_design_wsqrt * beta;
270
271 // vec gamma;
272 // SF.solve(gamma, X_features_wsqrt);
273 // T X_proj = X_features_wsqrt - T(X_design_wsqrt * gamma);
274
275 return {y_proj, X_proj};
276}

291static ModelFit wlm(const arma::mat& X, const arma::colvec& y, const arma::colvec& w = {}, const ModelDetail md = LOW, const double &rdf_offset = 0, LMWork *work = nullptr) {
292
293 ModelFit fit;
294
295 if( w.is_empty() ){
296 fit = lm( X, y, md, rdf_offset, work );
297 }else{
298 arma::colvec wsqrt = sqrt(w / mean(w));
299 fit = lm( X.each_col() % wsqrt, y % wsqrt, md, rdf_offset, work );
300
301 // Rescale residuals by weights afterward
302 // since input X and y are scaled before lm()
303 fit.residuals /= wsqrt;
304 }
305
306 return fit;
307}

320vector<ModelFit> lmFitFeatures_standard(const arma::vec &y, const arma::mat &X_design, const arma::mat &X_features, const vector<string> &ids, const arma::vec &weights = {}, const ModelDetail md = LOW, const int &nthreads = 1){
321
322 int n_covs = X_design.n_cols;
323
324 vector<ModelFit> fitList(X_features.n_cols, ModelFit());
325
326 #ifdef _OPENMP
327 // set threads
328 omp_set_num_threads(nthreads);
329 // disable nested parallelism
330 omp_set_max_active_levels(1);
331 #endif
332
333 #pragma omp parallel
334 {
335 // create design matrix with jth feature in the last column
336 // X = cbind(X_design, X_features[,0])
337 arma::mat X(X_design);
338 X.insert_cols(n_covs, X_features.col(0));
339
340 LMWork *work = new LMWork();
341
342 // iterate through responses
343 #pragma omp for
344 for(int j=0; j<X_features.n_cols; j++){
345 // Create design matrix with intercept as first column
346 X.col(n_covs) = X_features.col(j);
347
348 // linear regression
349 ModelFit fit = wlm(X, y, weights, md, 0, work);
350 fit.ID = ids[j];
351
352 // save result to list
353 fitList.at(j) = fit;
354 }
355 delete work;
356 }
357
358 return fitList;
359}
