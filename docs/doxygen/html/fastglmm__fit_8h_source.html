65template <typename T1, typename T2, typename T3>
66fastglmm<T1, T2, T3>::fastglmm(
67 const T1 &y,
68 const T2 &X,
69 const T3 &U,
70 const vec &s,
71 const vec &weights,
72 const vec &offset,
73 const string &family,
74 const ModelDetail md,
75 const double &tol,
76 const double &tol_eta,
77 const bool &returnUS):
78 family(family), returnUS(returnUS) {
79
80 shared_ptr<GLMFamily> fam = getGLMFamily( family );
81
82 // if Negative Binomial with unspecified theta
83 // estimate theta, and initialize with Poisson GLM
84 bool estimateTheta = family == "nb" ? true : false;
85 if( estimateTheta ){
86 this->family = "poisson/log";
87 }
88
89 checkResponse(y, this->family);
90
91 GLMWork *work = new GLMWork();
92 spectralDecomp<T3> dcmp;
93 vec eta_old;
94 T3 Z = scaleEachRow(U, sqrt(s));
95
96 // Initialize eta
97 // just need a rough starting value
98 ModelFitGLM fit_init = GLM(X, y, this->family, md, weights, offset, work, {}, 1e-2, 3);
99
100 int iter_in = 0;
101 double theta;
102
103 // PQL iterations
104 for(niter_pql=0; niter_pql<100; niter_pql++){
105
106 // if Negative Binomial with unspecified theta
107 if( estimateTheta ){
108 theta = nb_theta_ml(y, work->mu, y.n_elem, weights, {}, false);
109 fam->setOverdispersion( theta );
110 }
111
112 // update mu, eta, z, w, eta,
113 if( niter_pql == 0){
114 work->eta = work->eta + offset;
115 }else{
116 eta_old = work->eta;
117 work->eta = fit.fitted() + offset;
118
119 if( norm(work->eta - eta_old) < tol_eta){
120 break;
121 }
122 }
123
124 // mu <- family$linkinv(eta)
125 work->mu = fam->linkinv( work->eta );
126
127 // mu.eta.val <- family$mu.eta(eta)
128 work->gprime = fam->mu_eta( work->eta );
129
130 // zz <- eta + (y.orig - mu)/mu.eta.val - offset
131 work->z = (work->eta - offset) + (y - work->mu) / work->gprime;
132
133 // wz <- w * mu.eta.val^2/family$variance(mu)
134 work->w = pow(work->gprime,2) % (weights / fam->variance( work->mu ));
135
136 // wz <- wz / mean(wz)
137 work->w = work->w / mean(work->w);
138
139 // recompute U and s since work->w changed
140 dcmp.initWithIndicator(Z, work->w);
141
142 // fit fastlmm
143 fit = fastlmm(work->z, X, dcmp.get_vectors(), dcmp.get_values(), work->w, MAX);
144 fit.estimate_delta(-10, 10, tol);
145
146 // TODO
147 // 1) narrow delta search over time?
148 iter_in += fit.get_iter();
149 }
150
151 delete work;
152}
