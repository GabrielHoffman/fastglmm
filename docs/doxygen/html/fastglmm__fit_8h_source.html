76template <typename T1, typename T2, typename T3>
77fastglmm<T1, T2, T3>::fastglmm(
78 const T1 &y,
79 const T2 &X,
80 const spectralDecomp<T3> &dcmp,
81 const vec &weights,
82 const vec &offset,
83 const string &family,
84 const ModelDetail md,
85 const double &tol,
86 const double &tol_eta,
87 const int &maxit,
88 const double &delta,
89 const double &left,
90 const double &right,
91 const bool &returnUS):
92 y(y),
93 weights(weights),
94 dcmp(dcmp),
95 family(family),
96 returnUS(returnUS),
97 md(md)
98 {
99
100 fam = getGLMFamily( family );
101
102 // if Negative Binomial with unspecified theta
103 // estimate theta, and initialize with Poisson GLM
104 bool estimateTheta = family == "nb" ? true : false;
105 if( estimateTheta ){
106 this->family = "poisson/log";
107 }
108
109 checkResponse(y, this->family);
110
111 GLMWork *work = new GLMWork();
112 vec eta_old;
113
114 // Initialize eta
115 // just need a rough starting value
116 ModelFitGLM fit_init = GLM(X, y, this->family, LEAST, weights, offset, work, {}, 1e-2, 3);
117
118 int iter_in = 0;
119 double theta;
120 uvec idx_drop = find(weights == 0.0);
121 double n_active = weights.n_elem - idx_drop.n_elem;
122
123 // PQL iterations
124 for(niter_pql=0; niter_pql<maxit; niter_pql++){
125
126 // if Negative Binomial with unspecified theta
127 if( estimateTheta ){
128 theta = nb_theta_ml(y, work->mu, y.n_elem, weights, {}, false);
129 fam->setOverdispersion( theta );
130 }
131
132 // update mu, eta, z, w, eta,
133 if( niter_pql == 0){
134 work->eta = work->eta + offset;
135 }else{
136 eta_old = work->eta;
137 work->eta = fit.fitted() + offset;
138
139 // convergence criteria based on norm of eta change
140 if( norm(work->eta - eta_old) < tol_eta){
141 break;
142 }
143 }
144
145 // entries with zero weights have NAN value
146 work->eta.elem( idx_drop ).zeros();
147
148 // mu <- family$linkinv(eta)
149 work->mu = fam->linkinv( work->eta );
150
151 // mu.eta.val <- family$mu.eta(eta)
152 work->gprime = fam->mu_eta( work->eta );
153
154 // zz <- eta + (y.orig - mu)/mu.eta.val - offset
155 work->z = (work->eta - offset) + (y - work->mu) / work->gprime;
156
157 // wz <- w * mu.eta.val^2/family$variance(mu)
158 work->w = pow(work->gprime,2) % (weights / fam->variance( work->mu ));
159
160 // wz <- wz / mean(wz)
161 w_mean = sum(work->w) / n_active;
162 work->w = work->w / w_mean;
163
164 // recompute U and s since work->w changed
165 this->dcmp.reweight(work->w);
166
167 // fit fastlmm
168 fit = fastlmm(work->z, X, this->dcmp, work->w, LEAST);
169
170 if( delta > 0 ){
171 fit.eval_delta( delta );
172 }else{
173 fit.estimate_delta(left, right, tol);
174 }
175
176 // increment interation count
177 iter_in += fit.get_iter();
178 }
179
180 // Final fit with ModelDetail md
181 if( md > LEAST ){
182 // fit fastlmm
183 fit = fastlmm(work->z, X, this->dcmp, work->w, md);
184
185 if( delta > 0 ){
186 fit.eval_delta( delta );
187 }else{
188 fit.estimate_delta(left, right, tol);
189 }
190 }
191
192 if( estimateTheta ){
193 // update family to include estimated theta
194 this->family = "nb:" + to_string(theta);
195 }
196
197 // save GLM mu for use later
198 mu = this->fitted();
199
200 delete work;
201}

205template <typename T1, typename T2, typename T3>
206const vec fastglmm<T1, T2, T3>::residuals(){
207
208 // (y - mu) * sqrt(wts) / sqrt(fam$variance(mu))
209 return (y - mu) % sqrt(weights) / sqrt(fam->variance(mu));
210}

221template <typename T1, typename T2, typename T3>
222const vec fastglmm<T1, T2, T3>::devianceResiduals(){
223
224 // transform from residuals.glm
225 // d.res <- sqrt(pmax((object$family$dev.resids)(y, mu,
226 // wts), 0))
227 // ifelse(y > mu, d.res, -d.res)
228
229 // compute raw deviance residuals
230 vec dr = fam->dev_resids(y, mu, weights);
231
232 vec drMod = sqrt(pmax(dr, 0));
233 uvec idx = find(y <= mu);
234 drMod.elem(idx) = -1.0*drMod.elem(idx);
235
236 return drMod;
237}
