86template <typename T1, typename T2, typename T3>
87fastglmm<T1, T2, T3>::fastglmm(
88 const T1 &y,
89 const T2 &X,
90 const spectralDecomp<T3> &dcmp,
91 const vec &weights,
92 const vec &offset,
93 const string &family,
94 const ModelDetail md,
95 const double &tol,
96 const double &tol_eta,
97 const int &maxit,
98 const double &lambda,
99 const double &delta,
100 const double &left,
101 const double &right,
102 const bool &returnUS,
103 const bool &doCoxReid):
104 y(y),
105 X(X),
106 weights(weights),
107 offset(offset),
108 dcmp(dcmp),
109 family(family),
110 lambda(lambda),
111 returnUS(returnUS),
112 md(md)
113 {
114
115 fam = getGLMFamily( family );
116
117 // if Negative Binomial with unspecified theta
118 // estimate theta, and initialize with Poisson GLM
119 bool estimateTheta = family == "nb" ? true : false;
120 if( estimateTheta ){
121 this->family = "poisson/log";
122 }
123
124 checkResponse(y, this->family);
125
126 GLMWork *work = new GLMWork();
127 vec eta_old;
128
129 // Initialize eta
130 // just need a rough starting value
131 ModelFitGLM fit_init = GLM(X, y, this->family, LEAST, weights, offset, work, {}, 1e-2, 3, lambda);
132
133 int iter_in = 0;
134 double theta;
135 uvec idx_drop = find(weights == 0.0);
136 double n_active = weights.n_elem - idx_drop.n_elem;
137
138 CountTable ct;
139 if( estimateTheta ){
140 // Precompute lgamma() on each unique count
141 ct = CreateLUT(y, weights);
142 }
143
144 // PQL iterations
145 for(niter_pql=0; niter_pql<maxit; niter_pql++){
146
147 // if Negative Binomial with unspecified theta
148 if( estimateTheta ){
149 theta = nb_theta_ml(y, work->mu, y.n_elem, weights, X, doCoxReid, ct, -5, 20);
150 fam->setOverdispersion( theta );
151 }
152
153 // update mu, eta, z, w, eta,
154 if( niter_pql == 0){
155 work->eta = work->eta + offset;
156 }else{
157 eta_old = work->eta;
158 work->eta = fit.fitted() + offset;
159
160 // convergence criteria based on norm of eta change
161 if( norm(work->eta - eta_old) < tol_eta){
162 break;
163 }
164 }
165
166 // entries with zero weights have NAN value
167 work->eta.elem( idx_drop ).zeros();
168
169 // mu <- family$linkinv(eta)
170 work->mu = fam->linkinv( work->eta );
171
172 // mu.eta.val <- family$mu.eta(eta)
173 work->gprime = fam->mu_eta( work->eta );
174
175 // zz <- eta + (y.orig - mu)/mu.eta.val - offset
176 work->z = (work->eta - offset) + (y - work->mu) / work->gprime;
177
178 // wz <- w * mu.eta.val^2/family$variance(mu)
179 work->w = square(work->gprime) % (weights / fam->variance( work->mu ));
180
181 // wz <- wz / mean(wz)
182 w_mean = sum(work->w) / n_active;
183 work->w = work->w / w_mean;
184
185 // if model has nan values in z or w,
186 // it can't be fit
187 // so set beta values to nan
188 // and set isValid to false
189 if( work->z.has_nan() || work->w.has_nan() ){
190 fit.set_model_failure();
191 isValid = false;
192 break;
193 }
194
195 // recompute U and s since work->w changed
196 this->dcmp.reweight(work->w);
197
198 // fit fastlmm
199 fit = fastlmm(work->z, X, this->dcmp, work->w, LEAST, lambda);
200
201 if( delta > 0 ){
202 fit.eval_delta( delta );
203 }else{
204 fit.estimate_delta(left, right, tol);
205 }
206
207 // increment interation count
208 iter_in += fit.get_iter();
209 }
210
211 // Final fit with ModelDetail md
212 if( isValid && (md > LEAST) ){
213
214 // fit fastlmm
215 fit = fastlmm(work->z, X, this->dcmp, work->w, md, lambda);
216
217 if( delta > 0 ){
218 fit.eval_delta( delta );
219 }else{
220 fit.estimate_delta(left, right, tol);
221 }
222 }
223
224 if( estimateTheta ){
225 // update family to include estimated theta
226 this->family = "nb:" + to_string(theta);
227 }
228
229 // Use result of lmm and inverse link
230 // to get final value of mu
231 if( isValid ){
232 eta = fit.fitted() + offset;
233 mu = fam->linkinv(eta);
234 }else{
235 eta = vec(offset.n_elem, fill::value(datum::nan));
236 mu = vec(offset.n_elem, fill::value(datum::nan));
237 }
238
239 // Compute mean of mu
240 // use robust mean to avoid influence of outliers
241 // This can happen with many zero and a few large values
242 mu_mean = robust_mean(mu, 4);
243
244 eta_var = var(eta);
245
246 delete work;
247}

259template <typename T1, typename T2, typename T3>
260const vec fastglmm<T1, T2, T3>::residuals_pearson(){
261
262 // Pearson residuals
263 // (y - mu) * sqrt(wts) / sqrt(fam$variance(mu))
264 return (y - mu) % sqrt(weights) / sqrt(fam->variance(mu));
265}

275template <typename T1, typename T2, typename T3>
276const vec fastglmm<T1, T2, T3>::devianceResiduals(){
277
278 // transform from residuals.glm
279 // d.res <- sqrt(pmax((object$family$dev.resids)(y, mu,
280 // wts), 0))
281 // ifelse(y > mu, d.res, -d.res)
282
283 // compute raw deviance residuals
284 vec dr = fam->dev_resids(y, mu, weights);
285
286 vec drMod = sqrt(pmax(dr, 0));
287 uvec idx = find(y <= mu);
288 drMod.elem(idx) = -1.0*drMod.elem(idx);
289
290 return drMod;
291}

295template <typename T1, typename T2, typename T3>
296ModelFitGLMM fastglmm<T1, T2, T3>::get_result(){
297
298 ModelFitLMM res1 = fit.get_result(returnUS);
299 res1.set_w_mean( w_mean );
300
301 // if model is not valid, it failed before final calculations
302 // so set values to NAN matching ModelDetail
303 if( ! isValid ){
304 // number of coefs
305 int p = X.n_cols;
306 res1.coef = vec(p, fill::value(datum::nan));
307
308 switch( md ){
309 case MAX:
310 // res1.hatvalues = hatvalues();
311 case MOST:
312 res1.hatvalues.fill(datum::nan);
313 case HIGH:
314 res1.residuals.fill(datum::nan);
315 case MEDIUM:
316 res1.vcov = mat(p, p, fill::value(datum::nan));
317 case LOW:
318 res1.se = vec(p, fill::value(datum::nan));
319 res1.rdf = datum::nan;
320 case LEAST:
321 break;
322 }
323 }
324
325 ModelFitGLMM mf(res1, family, niter_pql);
326
327 mf.mu_mean = mu_mean;
328 mf.y_mean = mean(y);
329 mf.varFitted = eta_var;
330
331 if( md == MAX ){
332 mf.devianceResiduals = devianceResiduals();
333 }
334
335 if( md >= HIGH ){
336 // Respones residuals
337 mf.residuals = residuals();
338 }
339
340 return mf;
341}
