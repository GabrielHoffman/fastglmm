29class GLMFamily {

80class BinomialLogit :
81 public virtual GLMFamily {
82
83 public:
84 BinomialLogit() {}
85
86 ~BinomialLogit() override {}
87
88 vec link( const vec &mu) const override {
89 return log(mu / (1-mu));
90 }
91 vec linkinv( const vec &eta) const override {
92 return 1.0 / (1.0 + exp(-1.0*eta));
93 }
94 vec mu_eta( const vec &eta) const override {
95 vec v = exp(-1.0*eta);
96 return v / pow( 1.0 + v, 2);
97 }
98 vec variance( const vec &mu) const override {
99 return mu % (1.0 - mu);
100 }
101 vec dev_resids( const vec &y, const vec &mu, const vec &weights) const override {
102
103 // 2 * rwt[i] * (y_log_y(yi, mui) + y_log_y(1 - yi, 1 - mui));
104 return 2.0 * weights % (y_log_y(y, mu) + y_log_y(1.0 - y, 1.0 - mu));
105 }
106 vec initialize( const vec &y, const vec &weights) const override {
107 return (weights % y + 0.5)/(weights + 1.0);
108 }
109 bool estimateDispersion() const override {return false;}
110 string family() const override {return "BinomialLogit";}
111 bool isCountModel() const override { return false; }
112};

122class BinomialProbit :
123 public virtual GLMFamily {
124
125 public:
126 BinomialProbit(){}
127
128 ~BinomialProbit() override {}
129
130 vec link( const vec &mu) const override {
131 // qnorm(mu)
132 return qnorm(mu);
133 }
134 vec linkinv( const vec &eta) const override {
135 // pnorm(eta)
136 return normcdf( pmin(pmax(eta, -thresh), thresh) );
137 }
138 vec mu_eta( const vec &eta) const override {
139 // pmax(dnorm(eta), .Machine$double.eps)
140 return pmax(normpdf(eta), tol);
141 }
142 vec variance( const vec &mu) const override {
143 return mu % (1.0 - mu);
144 }
145 vec dev_resids( const vec &y, const vec &mu, const vec &weights) const override {
146
147 // 2 * rwt[i] * (y_log_y(yi, mui) + y_log_y(1 - yi, 1 - mui));
148 return 2.0 * weights % (y_log_y(y, mu) + y_log_y(1 - y, 1 - mu));
149 }
150 vec initialize( const vec &y, const vec &weights) const override {
151 return (weights % y + 0.5)/(weights + 1.0);
152 }
153 bool estimateDispersion() const override {return false;}
154 string family() const override {return "BinomialProbit";}
155 bool isCountModel() const override { return false; }
156
157 private:
158 double tol = 2.220446e-16;
159 // double thresh = -1.0 * R::qnorm( tol, 0.0, 1.0, 1, 0 );
160 double thresh = 8.125891; // -1*qnorm(.Machine$double.eps)
161};

164class PoissonLog :
165 public virtual GLMFamily {
166
167 public:
168 PoissonLog(){}
169
170 ~PoissonLog() override {}
171
172 vec link( const vec &mu) const override {
173 return log(mu);
174 }
175 vec linkinv( const vec &eta) const override {
176 // pmax(exp(eta), .Machine$double.eps)
177 return pmax(exp(eta), tol);
178 }
179 vec mu_eta( const vec &eta) const override {
180 // pmax(exp(eta), .Machine$double.eps)
181 return pmax(exp(eta), tol);
182 }
183 vec variance( const vec &mu) const override {
184 return mu;
185 }
186 vec dev_resids( const vec &y, const vec &mu, const vec &weights) const override {
187 // r <- mu * wt
188 // p <- which(y > 0)
189 // r[p] <- (wt * (y * log(y/mu) - (y - mu)))[p]
190 // 2 * r
191
192 vec res = mu % weights;
193 uvec idx = find(y > 0.0);
194 vec tmp = (weights % (y % log(y/mu) - (y - mu)));
195 res.elem(idx) = tmp.elem(idx);
196
197 return 2.0 * res;
198 }
199 vec initialize( const vec &y, const vec &weights) const override {
200 return y + 0.1;
201 }
202 bool estimateDispersion() const override {return false;}
203 string family() const override {return "PoissonLog";}
204 bool isCountModel() const override { return true; }
205
206 private:
207 double tol = 2.220446e-16;
208};

216class NB :
217 public virtual GLMFamily {
218
219 public:
220 NB(const double &theta) :
221 theta(theta) {}
222
223 ~NB() override {}
224
225 vec link( const vec &mu) const override {
226 return log(mu);
227 }
228 vec linkinv( const vec &eta) const override {
229 // pmax(exp(eta), .Machine$double.eps)
230 return pmax(exp(eta), tol);
231 }
232 vec mu_eta( const vec &eta) const override {
233 // pmax(exp(eta), .Machine$double.eps)
234 return pmax(exp(eta), tol);
235 }
236 vec variance( const vec &mu) const override {
237 return mu + pow(mu, 2) / theta;
238 }
239 vec dev_resids( const vec &y, const vec &mu, const vec &weights) const override {
240 // 2 * wt * (y * log(pmax(1, y)/mu) - (y + .Theta) * log((y + .Theta)/(mu + .Theta)))
241
242 return 2.0 * weights % (y % log(pmax(y, 1.0)/mu) - (y + theta) % log((y + theta)/(mu + theta)));
243 }
244 vec initialize( const vec &y, const vec &weights) const override {
245 // y + (y == 0)/6
246 return y + accu(y == 0) / 6.0;
247 }
248 bool estimateDispersion() const override {return true;}
249 string family() const override {return "NB";}
250 bool isCountModel() const override { return true; }
251
252 double theta;
253
254 private:
255 double tol = 2.220446e-16;
256
257};

264static shared_ptr<GLMFamily> getGLMFamily( const string &family){
265
266 shared_ptr<GLMFamily> fam;
267
268 if( family == "gaussian" || family == "gaussian/identity" ){
269 fam = shared_ptr<GLMFamily>(new GaussianIdentity());
270 }
271 else if( family == "binomial/logit" ){
272 fam = shared_ptr<GLMFamily>(new BinomialLogit());
273 }
274 else if( family == "binomial/probit" ){
275 fam = shared_ptr<GLMFamily>(new BinomialProbit());
276 }
277 else if( family == "poisson/log" ){
278 fam = shared_ptr<GLMFamily>(new PoissonLog());
279 }
280 else if( family == "quasibinomial/logit" ){
281 fam = shared_ptr<GLMFamily>(new QuasibinomialLogit());
282 }
283 else if( family == "quasipoisson/log" ){
284 fam = shared_ptr<GLMFamily>(new QuasipoissonLog());
285 }
286 else if( regex_search( family, regex("^nb:")) ){
287 string theta_str = regex_replace( family, regex("^nb:"), "");
288 double theta = atof(theta_str.c_str());
289 fam = shared_ptr<GLMFamily>(new NB(theta));
290 }else{
291 throw invalid_argument( "Invalid GLM family: " + family );
292 }
293
294 return fam;
295}
